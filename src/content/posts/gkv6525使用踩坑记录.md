国科微的注册函数回调是在一把锁内的所有无法执行复杂操作

```yaml

    eCode = uapi_svr_dispmng_register_callback(UAPI_SVR_DISPMNG_EVENT_ALL, lj_dispmng_event_process, NULL);
    if (!STATE_OK(eCode)) 
    {
        LOG_ERR("uapi_svr_dispmng_register_callback fail(ret=0x%#X)!\n", eCode);
        return STATE_CODE_INVALID_PARAM;
    }

```


回调函数是

```yaml
VOID lj_dispmng_event_process(uapi_svr_dispmng_event event, VOID* msg, UINT32 len UNUSED, VOID* priv_data UNUSED)
{
    uapi_svr_dispmng_event_mode_changed *mode_event = msg;
    uapi_svr_dispmng_event_hdr_changed *hdr_event = msg;
    uapi_svr_dispmng_event_hdmitx_event *hdmi_event = msg;
    T_DisplayNode *ptNode = NULL;

    if(msg == NULL)
    {
        syslog("DISPMNG event msg is NULL POINTER\n");
        return;
    }

    switch (event) {
        case UAPI_SVR_DISPMNG_EVENT_MODE_CHANGED:
            syslog("DISPMNG event (type:\"DISPLAY_MODE_CHANGE\", id:%u, old_mode:%s(%u), new_mode:%s(%u))\n",
                mode_event->display_id, get_disp_vfmt_str(mode_event->old_mode.format),mode_event->old_mode.format,\
                 get_disp_vfmt_str(mode_event->new_mode.format),mode_event->new_mode.format);
            break;
        case UAPI_SVR_DISPMNG_EVENT_HDR_CHANGED:
            syslog("DISPMNG event (type:\"HDR_OUTPUT_CHANGE\", id:%u, old_hdr_type:%u, new_hdr_type:%u)\n",
                hdr_event->display_id, hdr_event->old_hdr, hdr_event->new_hdr);
            break;
        case UAPI_SVR_DISPMNG_EVENT_HOTPLUG_IN:
        {
            syslog("DISPMNG event (type:\"HDMI HOTPLUG IN\", disp:%u, hdmitx:%u)\n",
                hdmi_event->display_id, hdmi_event->hdmitx_id);
            ptNode = SELECT_DISPLAY_NODE(hdmi_event->display_id, pMasterNode, pSlaveNode);
            if(ptNode != NULL)
            {
                ptNode->bIsConnected = TRUE;
                if(ptNode->tParams.id == UAPI_SVR_DISPMNG_DISPLAY_ID_SLAVE)
                {
                    // set_display_bit_depth(UAPI_SVR_DISPMNG_DISPLAY_ID_SLAVE, UAPI_SVR_DISPMNG_PIXEL_BIT_DEPTH_8BIT);
                    // set_display_pixel_format(UAPI_SVR_DISPMNG_DISPLAY_ID_SLAVE, UAPI_SVR_DISPMNG_PIXEL_FORMAT_YUV422);
                    // set_display_format(UAPI_SVR_DISPMNG_DISPLAY_ID_SLAVE, ptNode->tParams.format);
                }
                else if(ptNode->tParams.id == UAPI_SVR_DISPMNG_DISPLAY_ID_MASTER)
                {
                    
                    // set_display_native_mode(UAPI_SVR_DISPMNG_DISPLAY_ID_MASTER);
                }
            }
            break;
        }
        case UAPI_SVR_DISPMNG_EVENT_HOTPLUG_OUT:
        {
            syslog("DISPMNG event (type:\"HDMI HOTPLUG OUT\", disp:%u, hdmitx:%u)\n",
                hdmi_event->display_id, hdmi_event->hdmitx_id);
            ptNode = SELECT_DISPLAY_NODE(hdmi_event->display_id, pMasterNode, pSlaveNode);
            if(ptNode != NULL)
            {
                ptNode->bIsConnected = FALSE;
            }

            if(ptNode->tParams.id == UAPI_SVR_DISPMNG_DISPLAY_ID_MASTER)
            {
                ptNode->bNeedUpdate = TRUE;
            }
            // uapi_svr_dispmng_get_display_mode(hdmi_event->display_id, &mode);  
            // mode.vic = UAPI_SVR_DISPMNG_DISPLAY_MODE_ADVANCED; //强制显示
            // uapi_svr_dispmng_set_display_mode(hdmi_event->display_id, &mode); 
            // LOG_ERR("uapi_svr_dispmng_set_display_mode ADVANCED.\n");

            // uapi_svr_dispmng_get_display_status(hdmi_event->display_id, &status);
            // parse_status_and_print(&status);
            break;
        }

        case UAPI_SVR_DISPMNG_EVENT_HDCP_CAP_CHANGE:
        case UAPI_SVR_DISPMNG_EVENT_HDCP_FAIL:
        case UAPI_SVR_DISPMNG_EVENT_HDCP_SUCCESS:
        case UAPI_SVR_DISPMNG_EVENT_RSEN_CONNECT:
        case UAPI_SVR_DISPMNG_EVENT_RSEN_DISCONNECT:
        case UAPI_SVR_DISPMNG_EVENT_LINK_TRAIN_FAIL:
            syslog("HDMI event (type:%u, display_id:%u, hdmi_id:%u)\n",
                event, hdmi_event->display_id, hdmi_event->hdmitx_id);
            break;
        default:
            syslog("Unknown event (type:%u)\n", event);
    }
}
```
